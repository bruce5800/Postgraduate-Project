% A rule-delimited annotation block. Deliberately built from LaTeX primitives only
% (no tcolorbox/mdframed/framed) so it compiles on a basic TeX Live.
\newcommand{\revsavedcolor}{revShould}
\newenvironment{revbox}[2]{%
  \renewcommand{\revsavedcolor}{#1}%
  \par\medskip\noindent
  {\color{#1}\rule{\linewidth}{1pt}}\par\nobreak\vspace{2pt}%
  \noindent{\small\bfseries\sffamily\color{#1}#2}\par\nobreak\vspace{2pt}%
  \small\color{#1!85!black}\ignorespaces
}{%
  \par\vspace{2pt}\noindent{\color{\revsavedcolor}\rule{\linewidth}{0.4pt}}\par\medskip
}
